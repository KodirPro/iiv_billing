## Project setup

```bash

# PostgreSQL installing
sudo apt install curl ca-certificates -y
sudo install -d /usr/share/postgresql-common/pgdg
sudo curl -o /usr/share/postgresql-common/pgdg/apt.postgresql.org.asc --fail https://www.postgresql.org/media/keys/ACCC4CF8.asc
sudo sh -c 'echo "deb [signed-by=/usr/share/postgresql-common/pgdg/apt.postgresql.org.asc] https://apt.postgresql.org/pub/repos/apt $(lsb_release -cs)-pgdg main" > /etc/apt/sources.list.d/pgdg.list'
sudo apt update
sudo apt install postgresql -y
sudo systemctl enable postgresql
sudo systemctl restart postgresql
sudo -u postgres psql -c "ALTER USER postgres WITH ENCRYPTED PASSWORD '008';"


# NodeJS installing
sudo apt install curl git -y
curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/$(curl -s https://api.github.com/repos/nvm-sh/nvm/releases/latest | grep -oP '"tag_name": "\K(.*)(?=")')/install.sh | bash
source ~/.bashrc
nvm ls-remote
nvm install lts/iron
nvm alias default lts/iron
nvm use default


# PNPM installing
sudo apt install curl -y
curl -fsSL https://get.pnpm.io/install.sh | sh -
echo -e "\n# pnpm alias\nalias pm=pnpm;\nalias px=\"pnpm dlx\";\n" >> ~/.bashrc
source ~/.bashrc


# prisma.schema setup
generator client {
  provider = "prisma-client-js"
  output   = "../node_modules/.prisma/client"
}


# Build and create binary file
nvm use 20
pm add -g terser nexe
pm i
px prisma generate
pm build
pm binary

```

## Project Deploying

```bash
# Service configuring
bash ./configure.sh


# Service control
sudo systemctl enable billing.service
sudo systemctl start billing.service

sudo systemctl stop billing.service
sudo systemctl disable billing.service

sudo systemctl status billing.service
sudo systemctl restart billing.service


# Service logs
sudo journalctl -u billing.service
sudo journalctl -u billing.service -n 50
sudo journalctl -u billing.service -f
```
